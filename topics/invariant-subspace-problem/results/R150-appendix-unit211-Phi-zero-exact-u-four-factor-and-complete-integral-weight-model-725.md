$$

$$
\operatorname{ord}_{u=0}F=4.
\tag{1.1}
$$
因而存在唯一 $F_0\in\mathbb Q[u,Q]$ 满足
$$
F=u^4F_0,\qquad \deg_uF_0=725.
\tag{1.2}

$$
\deg F_0(u,2)=\deg\overline{F_0}(u,2)=725,\qquad
[u^{725}]\overline{F_0}(u,2)=702.
\tag{1.3}
