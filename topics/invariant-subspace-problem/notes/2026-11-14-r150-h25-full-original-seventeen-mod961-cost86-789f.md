\[

\beta_i=A\widehat\kappa_i/31\pmod{31}.
\]

\[
J_v-HC=(s-2)(J_r-c_1J_{\beta_1}-c_2J_{\beta_2})-HC.
\]

\[
\boxed{X_j=(s-2)(x_{0,j}+c_{1j}\widehat\kappa_1+c_{2j}\widehat\kappa_2)

\]
